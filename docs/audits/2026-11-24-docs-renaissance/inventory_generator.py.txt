    result = {}

        rel = x if isinstance(x, str) else x['path']
        result[(root / 'docs' / rel).resolve().relative_to(root).as_posix()] = None if isinstance(x, str) else x.get('reason')
    return result

        row['source_locators'] = row.get('source_locators', row.get('evidence', list(row.get('source_sha256', reviewed.get('source_sha256', {})))))

            row['inspection'] = 'Bounded native identity/capability and pair-participant table checks only. A explicitly retained wider scientific/material-behavior claims at original evidence scope; this is not a complete source or runtime requalification.'
        deep_reviews[path].append({**row, 'record': name})
